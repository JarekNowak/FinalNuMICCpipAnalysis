\begin{frame}{Event selection: cut sequence}
  \footnotesize
  Topological CC$1\mu1\pi^\pm Xp$ selection applied sequentially (cosmics $\to$ PID $\to$ kinematics):
  \begin{columns}[T]\column{.5\textwidth}
  \begin{enumerate}\setlength\itemsep{1pt}
    \item[\textbf{1.}] \textbf{NeutrinoSlice}: $\ge1$ track, software trigger.
    \item[\textbf{2.}] \textbf{InFiducialVol}: vertex inside active TPC FV.
    \item[\textbf{3.}] \textbf{Topological}: Pandora score $>0.67$ (EXT $\times33$ down).
    \item[\textbf{4.}] \textbf{MuonCandidate}: passes $\mu$ PID; longest track $=\mu$.
    \item[\textbf{5.}] \textbf{ContainedPion}: 1 contained track with $\pi$ PID.
  \end{enumerate}
  \column{.5\textwidth}
  \begin{enumerate}\setlength\itemsep{1pt}
    \item[\textbf{6--7.}] \textbf{$\mu$/$\pi$ In3Planes}: hits on U, V, and Y planes.
    \item[\textbf{8.}] \textbf{ShowerCut}: veto EM showers ($\pi^0$ rejection).
    \item[\textbf{9.}] \textbf{OpeningAngle}: $\theta_{\mu\pi}<2.6$\,rad ($149^\circ$) cosmic veto.
    \item[\textbf{10.}] \textbf{Nonprotons}: $<4$ non-proton tracks, $<5$ total.
  \end{enumerate}
  \end{columns}
  \vspace{1mm}
\bigskip
  {\large \textbf{Final (measured phase space): efficiency $17.5\%$, purity $55.7\%$} }
\end{frame}

\begin{frame}{Momentum reconstruction: range, MCS and proton KE}
  \footnotesize
  Contained tracks use \alert{range-based momentum} via $dE/dx$; exiting muons use \alert{multiple Coulomb scattering (MCS)}.
  \begin{center}\scriptsize
  \begin{tabular}{llll}
  \toprule
  Particle & Estimator & PeLEE branch & Note\\\midrule
  $\mu$ (contained) & range momentum & \texttt{trk\_range\_muon\_mom\_v} & \alert{37\%} of selected\\
  $\mu$ (exiting)   & MCS momentum   & \texttt{trk\_mcs\_muon\_mom\_v}   & \alert{63\%}, range invalid\\
  $\pi$             & range momentum, \alert{$\mu$ hypothesis} & \texttt{trk\_range\_muon\_mom\_v} & $m_\mu \approx m_\pi$\\
  $p$               & range \alert{KE}, $p$ hypothesis & \texttt{trk\_energy\_proton\_v} & $\to$ momentum $|\vec p_p|$\\
  \bottomrule
  \end{tabular}
  \end{center}
  {\scriptsize $p_\mu$ is a \alert{hybrid} observable; MCS-dominated because most energetic muons leave the active volume.}\\[1mm]
  \begin{block}{Pion range momentum ($\mu$ hypothesis)}\scriptsize
  Close masses ($m_\mu=105.7$\,MeV, $m_\pi=139.6$\,MeV) make muon range momentum an accurate proxy for stopping pions.\\[1mm]
  Directly comparable to true $p_\pi$, preserving a near-diagonal response matrix. Requires contained pions.
  \end{block}
\end{frame}

\begin{frame}{Why the pion momentum cannot be measured well}
  \centering
  \begin{columns}[c]
  \column{.33\textwidth}\centering
    \includegraphics[width=\linewidth,height=.38\textheight,keepaspectratio]{ppi_saturation}\\[0.5mm]
    {\scriptsize \alert{Saturates} above $\sim\!0.35$\,GeV/$c$}
  \column{.33\textwidth}\centering
    \includegraphics[width=\linewidth,height=.38\textheight,keepaspectratio]{resolution_ppi}\\[0.5mm]
    {\scriptsize \alert{Systematic under-reconstruction}}
  \column{.33\textwidth}\centering
    \includegraphics[width=\linewidth,height=.38\textheight,keepaspectratio]{fw_resp_ppi}\\[0.5mm]
    {\scriptsize Reco piles into \alert{lowest bin}}
  \end{columns}
  \vspace{3mm}
  \begin{itemize}\small\setlength\itemsep{3pt}
    \item \textbf{Hadronic re-interactions}: Unlike muons, charged pions scatter or get absorbed in argon before stopping $\Rightarrow$ premature track endings cause severe momentum underestimation.
    \item \textbf{Fundamental detector limit}: Tagging cleanly stopping pions via BDT (or MC truth) reaches a ceiling without resolving the migration.
    \item \textbf{Mitigation}: Pion momentum is reported in a robust \alert{two-bin scheme}; all other kinematic observables remain unaffected.
  \end{itemize}
\end{frame}

\begin{frame}{Pion measurement reduced to two bins}
  \centering
  \begin{columns}[T]
    \column{.31\textwidth}\centering\includegraphics[width=\textwidth]{ppi2bin_AC_FHC}
    \column{.31\textwidth}\centering\includegraphics[width=\textwidth]{ppi2bin_AC_RHC}
    \column{.31\textwidth}\centering\includegraphics[width=\textwidth]{ppi2bin_AC_COMB}
  \end{columns}
  \vspace{2mm}
  \begin{itemize}\small\setlength\itemsep{3pt}
    \item \textbf{Inter-bin correlation}: $A_C$ carries negative off-diagonal entries (borrowing between bins) due to strong bin-to-bin migration.
    \item \textbf{Configuration consistency}: all three now regularise similarly (diagonals $0.59, 0.51, 0.51$); the off-diagonal grows with exposure ($-0.38, -0.43, -0.53$). The combined diagonal was $0.34$ before the regularisation fix.
    \item \textbf{Normalisation}: unfolded/truth is $1.15/1.12$ (FHC), $0.98/1.14$ (RHC), $1.06/1.14$ (comb). The uniform $1.13$ quoted previously was partly an artefact of the over-regularised response.
    \item \textbf{Response}: migration diagonals \alert{$91/76\%$} (five-bin: $96/35/21/10/6\%$), both
      above the $68\%$ criterion; an exhaustive scan finds \alert{no 3- or 4-bin scheme clears it}.
  \end{itemize}
\end{frame}

\begin{frame}{FHC: unfolding}

\begin{frame}{The six proton-tagged observables}
  \footnotesize
  Constructed from $\mu$, $\pi^\pm$, and leading $p$ four-vectors ($\vec p^{\,\mathrm{had}}=\vec p_\pi+\vec p_p$; transverse components w.r.t.\ beam $\hat z$):
  \vspace{-1mm}
  \begin{columns}[T]\column{.52\textwidth}
  \begin{block}{Hadronic mass (probes $\Delta / N^\ast$)}\scriptsize
  \begin{align*}
  W_{\pi p} &= \sqrt{(E_\pi+E_p)^2-|\vec p_\pi+\vec p_p|^2}\\
  W_\mathrm{had} &= \sqrt{m_p^2+2m_p(E_\mathrm{cal}-E_\mu)-Q^2}
  \end{align*}
  $W_{\pi p}$ is \alert{directly measured} (no $\nu$ beam assumption). $W_\mathrm{had}$ is calorimetric ($E_\mathrm{cal}=E_\mu+E_\pi+T_p$).\\
  Physical threshold: $W_{\pi p}\ge m_\pi+m_p=1.08$\,GeV/$c^2$.
  \end{block}
  \column{.48\textwidth}
  \begin{block}{TKI (probes Fermi motion \& FSI)}\scriptsize
  \begin{align*}
  \delta p_T &= |\vec p_T^{\,\mu}+\vec p_T^{\,\mathrm{had}}|\\
  \delta\alpha_T &= \angle\big(-\vec p_T^{\,\mu},\ \vec p_T^{\,\mu}+\vec p_T^{\,\mathrm{had}}\big)\\
  \delta\phi_T &= \angle\big(-\vec p_T^{\,\mu},\ \vec p_T^{\,\mathrm{had}}\big)\\
  p_n &= \sqrt{\delta p_T^2+p_L^2}
  \end{align*}
  $\delta p_T \to 0$ without nuclear effects; \alert{Fermi motion broadens} $\delta p_T$ and \alert{FSI shifts $\delta\alpha_T \to 180^\circ$}. $p_n$ infers initial nucleon momentum ($E_b=24.78$\,MeV argon binding energy).
  \end{block}
  \end{columns}
  \vspace{1mm}
  {\scriptsize Measured in six equal-population bins. Requires contained proton with $p_p > 0.3$\,GeV/$c$.}
\end{frame}

\begin{frame}{Backup: historical note; pion-momentum estimator}
  \footnotesize
  An earlier iteration reconstructed $p_\pi$ using the \alert{proton-hypothesis range KE} (\texttt{trk\_energy\_proton\_v}):
  \begin{itemize}\setlength\itemsep{2pt}
    \item \textbf{Mismatch}: Kinetic energy was mapped directly onto a momentum axis, distorting the migration matrix.
    \item \textbf{Consequence}: A true $0.20$\,GeV/$c$ pion gave $\mathrm{KE}\approx0.10$\,GeV (shifted two bins low); low-momentum pions fell below the first reco bin.
    \item \textbf{Resolution}: Switching to \texttt{trk\_range\_muon\_mom\_v} exploits $m_\mu \approx m_\pi$ and restores a near-diagonal response matrix.
  \end{itemize}
  \vspace{1mm}
  {\scriptsize Note: Proton-hypothesis range KE remains the correct estimator for \alert{protons} in the W/TKI subsample ($|\vec p_p|=\sqrt{T_p^2+2T_pm_p}$).}
\end{frame}

\begin{frame}{Backup: W/TKI observable definitions}
  \small Reconstructed from $\mu$, $\pi^\pm$, and leading $p$ four-vectors ($\vec p^{\,\mathrm{had}}=\vec p_\pi+\vec p_p$):

  W_{\pi p} &= \sqrt{(E_\pi+E_p)^2-|\vec p_\pi+\vec p_p|^2}\quad\text{(direct, beam-independent)}\\
  W_\mathrm{had} &= \sqrt{m_p^2+2m_p(E_\mathrm{cal}-E_\mu)-Q^2}\quad\text{(calorimetric)}\\

  \delta\alpha_T,\ \delta\phi_T\ \text{(TKI angles)},\quad p_n=\sqrt{\delta p_T^2+p_L^2}
  \end{align*}
