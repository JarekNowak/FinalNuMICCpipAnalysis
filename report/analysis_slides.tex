\begin{frame}{Event selection: cut sequence}
  \footnotesize
  A CC$1\mu1\pi^\pm Xp$ topological selection: one muon, exactly one contained charged
  pion, any number of protons. Applied in order (cosmic rejection first, then PID, then
  kinematic cleanup):
  \begin{columns}[T]\column{.5\textwidth}
  \begin{enumerate}\setlength\itemsep{1pt}
    \item[\textbf{1.}] \textbf{NeutrinoSlice} --- $\ge1$ track, software trigger.
    \item[\textbf{2.}] \textbf{InFiducialVol} --- SCE-corrected $\nu$ vertex in FV.
    \item[\textbf{3.}] \textbf{Topological} --- Pandora score $>0.67$; primary cosmic
      rejection (EXT $\times33$ down).
    \item[\textbf{4.}] \textbf{MuonCandidate} --- $\ge1$ primary track passes $\mu$ PID;
      longest $=\mu$.
    \item[\textbf{5.}] \textbf{ContainedPion} --- exactly one other track passes $\pi$
      PID, endpoint contained; highest LLR $=\pi$.
  \end{enumerate}
  \column{.5\textwidth}
  \begin{enumerate}\setlength\itemsep{1pt}
    \item[\textbf{6--7.}] \textbf{$\mu$/$\pi$ In3Planes} --- hit on each of U, V, Y.
    \item[\textbf{8.}] \textbf{ShowerCut} --- veto primary EM showers ($\pi^0$ rejection;
      one soft shower tolerated).
    \item[\textbf{9.}] \textbf{OpeningAngle} --- $\theta_{\mu\pi}<2.6$~rad ($149^\circ$);
      rejects large-angle cosmic/dirt.
    \item[\textbf{10.}] \textbf{Nonprotons} (final) --- $<4$ non-proton-like tracks
      (LLR$>0.1$) and $<5$ total primary tracks.
  \end{enumerate}
  \end{columns}
  \vspace{1mm}
  {\scriptsize Per-run POT scaling; EXT (beam-off) supplies data-driven cosmics; dirt from
  the GENIE NuMI overlay. \textbf{Final: efficiency $15.6\%$, purity $55\%$} (measured
  phase space).}
\end{frame}

\begin{frame}{Momentum reconstruction: range, MCS and the proton range-KE}
  \footnotesize
  All three tracks are \alert{range-based} where possible: a contained track's length maps
  onto momentum through the particle's $dE/dx$, which is far more precise than calorimetry
  at these energies.
  \begin{center}\scriptsize
  \begin{tabular}{llll}
  \toprule
  Particle & Estimator & PeLEE branch & Note\\\midrule
  $\mu$ (contained) & range momentum & \texttt{trk\_range\_muon\_mom\_v} & \alert{37\%} of selected\\
  $\mu$ (exiting)   & MCS momentum   & \texttt{trk\_mcs\_muon\_mom\_v}   & \alert{63\%} --- range invalid\\
  $\pi$             & range momentum, \alert{$\mu$ hypothesis} & \texttt{trk\_range\_muon\_mom\_v} & $m_\mu$ close to $m_\pi$\\
  $p$               & range \alert{KE}, $p$ hypothesis & \texttt{trk\_energy\_proton\_v} & $\to$ momentum, see below\\
  \bottomrule
  \end{tabular}
  \end{center}
  {\scriptsize $p_\mu$ is therefore a \alert{hybrid} quantity (\texttt{..\_mom\_reco}
  switches on muon containment) and is \alert{MCS-dominated}: NuMI muons are energetic and
  the TPC is small, so most leave it.}\\[1mm]
  \begin{columns}[T]\column{.52\textwidth}
  \begin{block}{Proton: range KE $\to$ momentum}\scriptsize
  The proton branch gives a \emph{kinetic energy}, so it is converted:
  \[ |\vec p_p|=\sqrt{T_p^{2}+2\,T_p\,m_p} \]
  Threshold $|\vec p_p|>0.3$~GeV/$c$ ($T_p\gtrsim48$~MeV) --- below this the proton is too
  short to track. Performance: \alert{$-9\%$ bias, $21\%$ resolution}.
  \end{block}
  \column{.48\textwidth}
  \begin{block}{Why the pion uses the $\mu$ hypothesis}\scriptsize
  $m_\mu=105.7$~MeV, $m_\pi=139.6$~MeV: the masses are close, so the range$\to$momentum
  curve is a good proxy for a stopping pion.\\[1mm]
  It also returns a \alert{momentum}, directly comparable to the true $p_\pi$ it is binned
  against --- which keeps the migration matrix near-diagonal.\\[1mm]
  Valid only for \alert{contained} tracks; the selection requires pion containment.
  \end{block}
  \end{columns}
\end{frame}

\begin{frame}{FHC: unfolding}
  \begin{columns}[T]\column{.55\textwidth}
  \begin{itemize}\footnotesize\setlength\itemsep{3pt}
    \item The background-subtracted reco spectrum is \alert{not} the cross section: it
      still carries the detector response and the selection efficiency.

\begin{frame}{Why $\theta_\mu$: the $A_C$ smearing penalises the forward peak}
  \begin{columns}[T]\column{.52\textwidth}
  \centering\scriptsize
  \begin{tabular}{llcccc}
  \toprule
  Beam & Var & data & tune & GENIE & NuWro\\\midrule
  FHC & $\cos\theta_\mu$ & 0.82 & 0.67 & 0.27 & 0.30\\
  FHC & $\theta_\mu$     & 1.00 & 0.87 & 0.63 & 0.92\\\midrule
  RHC & $\cos\theta_\mu$ & 0.66 & 0.63 & 0.23 & 0.26\\
  RHC & $\theta_\mu$     & 0.83 & 0.82 & 0.33 & 0.42\\
  \bottomrule
  \end{tabular}\\[2mm]
  {\scriptsize $A_C$ suppression (smeared/truth):\\
  FHC $\cos\theta_\mu$ \alert{0.33} $\to$ $\theta_\mu$ \alert{0.76} \quad($p_\mu$ 0.64)\\
  RHC $\cos\theta_\mu$ 0.34 $\to$ $\theta_\mu$ 0.49 \quad($p_\mu$ 0.72)}
  \column{.48\textwidth}
  \begin{itemize}\footnotesize
    \item $\cos\theta_\mu$ is sharply forward-peaked; the regularised measurement cannot
      resolve it, so $A_C$ smears it away --- penalising the \alert{peaked generators}
      far more than the smoother tune.
    \item In $\theta_\mu$ the $\sin\theta$ Jacobian flattens the peak $\Rightarrow$
      suppression $0.33\!\to\!0.76$, generators land \alert{on} the data.
    \item Not a model disagreement: truth-level generator totals agree; it is a
      \alert{variable-choice artefact}.
    \item RHC improves less (poorer angular response, stronger regularisation).
      $\theta_\pi$ $\approx$ $\cos\theta_\pi$ (pion never sharply peaked).
  \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Part II --- Proton-tagged subsample: hadronic mass \& TKI}

\begin{frame}{The six proton-tagged observables}
  \footnotesize
  Built from the muon, charged-pion and leading-proton four-vectors
  ($\vec p^{\,\mathrm{had}}=\vec p_\pi+\vec p_p$, transverse components w.r.t.\ the beam
  $\hat z$; framework STV tool):
  \vspace{-1mm}
  \begin{columns}[T]\column{.52\textwidth}
  \begin{block}{Hadronic mass --- probes $\Delta/N^\ast$}\scriptsize
  \begin{align*}
  W_{\pi p} &= \sqrt{(E_\pi+E_p)^2-|\vec p_\pi+\vec p_p|^2}\\
  W_\mathrm{had} &= \sqrt{m_p^2+2m_p(E_\mathrm{cal}-E_\mu)-Q^2}
  \end{align*}
  $W_{\pi p}$ is \alert{directly measured} --- no assumption on the $\nu$ direction or
  energy. $W_\mathrm{had}$ is calorimetric, $E_\mathrm{cal}=E_\mu+E_\pi+T_p$.\\[1mm]
  Threshold: $W_{\pi p}\ge m_\pi+m_p=1.08$~GeV/$c^2$.
  \end{block}
  \column{.48\textwidth}
  \begin{block}{TKI --- probes Fermi motion \& FSI}\scriptsize
  \begin{align*}
  \delta p_T &= |\vec p_T^{\,\mu}+\vec p_T^{\,\mathrm{had}}|\\
  \delta\alpha_T &= \angle\big(-\vec p_T^{\,\mu},\ \vec p_T^{\,\mu}+\vec p_T^{\,\mathrm{had}}\big)\\
  \delta\phi_T &= \angle\big(-\vec p_T^{\,\mu},\ \vec p_T^{\,\mathrm{had}}\big)\\
  p_n &= \sqrt{\delta p_T^2+p_L^2}
  \end{align*}
  With no nuclear effects $\delta p_T\!\to\!0$; \alert{Fermi motion broadens} it and
  \alert{FSI drive $\delta\alpha_T$ toward $180^\circ$}. $p_n$ infers the initial-state
  nucleon momentum ($E_b=24.78$~MeV argon removal energy enters $p_L$).
  \end{block}
  \end{columns}
  \vspace{1mm}
  {\scriptsize Each is measured in six equal-population bins; the proton is required
  contained with $|\vec p_p|>0.3$~GeV/$c$.}
\end{frame}

\section{Multi-pion}
\begin{frame}{Part III --- Multi-pion: exclusive CC$\,1\mu2\pi$ and CC$\,1\mu3\pi$}
  \footnotesize
  New exclusive channels, building on the work of \textbf{Tusharadri Mahmud} (MSc) and
  \textbf{Linhui Gu}. Thin subselections of the inclusive selection: same muon, but
  \emph{exactly} 2 or 3 charged pions (any number of protons). They probe the resonant/DIS
  transition and the intranuclear pion cascade, and are a background to the single-pion
  signal.

  Wire-gap cuts        & on & off\\

  \end{tabular}

  \end{columns}
\end{frame}

\begin{frame}{Multi-pion: dedicated pion-ID BDT}
  \begin{columns}[T]\column{.54\textwidth}
  \footnotesize
  \begin{itemize}\setlength\itemsep{2pt}
    \item Inherited loose LLR cut replaced by a \alert{dedicated BDT} (TMVA gradient
      boost), trained to separate true charged pions from the tracks polluting the
      multi-$\pi$ sample. \alert{ROC $=0.81$}.
    \item Inputs: pion/proton LLR, Bragg $p/\mu$/MIP likelihoods, track score, length,
      vertex distance --- no explicit momentum.
    \item $2\pi$ uses \alert{two momentum-specialised BDTs} (split at $20$ cm track
      length, tighter soft-pion cut to protect the exact-multiplicity purity);
      $3\pi$ keeps a \alert{single BDT} (its pions are genuinely soft).
    \item Trialled on the \alert{inclusive} selection it \alert{hurts}
      (eff $16.7\!\to\!18.9\%$ but purity $60.3\!\to\!38.8\%$) $\Rightarrow$ inclusive
      keeps its LLR cut.
  \end{itemize}
  \column{.46\textwidth}
  \begin{block}{Residual background (truth-matched)}\footnotesize
  \begin{tabular}{lr}
  \toprule Counted ``pion'' really is & \%\\\midrule
  real out-of-topology $\pi^\pm$ & 44\\
  proton mis-ID & 31\\
  muon & 12\\
  other & 13\\
  \bottomrule
  \end{tabular}
  \end{block}
  {\scriptsize Proton contamination is the \alert{irreducible} core --- a proton ranging
  out like a pion is genuinely ambiguous; adding a Bragg veto on top of the BDT did
  \emph{not} reduce it and only cost efficiency.}\\[1mm]
  {\scriptsize Documented in a standalone technote; differential $2\pi$ measurement +
  total $\sigma$ for $3\pi$ are the targets.}
\end{columns}
\end{frame}

  \begin{itemize}
    \item \alert{Inclusive}: differential CC$1\mu1\pi^{\pm}$ cross sections for FHC, RHC
      and combined, per-run POT scaling, Wiener-SVD unfolding; total flux-averaged
      $\sigma$ agrees with GENIE/GiBUU/NEUT/NuWro within $\sim1\sigma$.
    \item \alert{Proton-tagged}: new hadronic-mass ($W_{\pi p}$, $W_\mathrm{had}$) and
      TKI ($\delta p_T$, $\delta\alpha_T$, $\delta\phi_T$, $p_n$) observables, plus the
      five inclusive kinematic observables re-measured. \alert{FHC closure validated} at

    \item Blind analysis: fake data throughout; sidebands (CC0$\pi$, multi-$\pi$, $\pi^0$,
      cosmic) validated in all three beams, data/$\nu$MC $\simeq1.00$, ready to constrain
      the background before unblinding.
    \item \alert{Multi-pion}: an analysis of the exclusive $2\pi$ and $3\pi$ channels has
      started, building on the work of Tusharadri Mahmud (MSc) and Linhui Gu.
  \end{itemize}

\end{frame}

\begin{frame}{Backup: history --- the superseded pion-momentum estimator}
  \footnotesize
  An earlier version of this analysis reconstructed the \alert{pion} momentum from the
  \alert{proton-hypothesis range kinetic energy} (\texttt{trk\_energy\_proton\_v}) rather
  than the muon-hypothesis range momentum used now.
  \begin{itemize}\setlength\itemsep{2pt}
    \item That is a \alert{kinetic energy}, while the true axis it was binned against
      (with identical bin edges) is a \alert{momentum} --- two different physical
      quantities.
    \item Consequence: the migration matrix was \alert{systematically shifted} rather than
      near-diagonal. A true $0.2$~GeV/$c$ pion gives $KE\approx0.104$~GeV, \alert{two bins
      low}; a true $0.175$~GeV/$c$ pion falls \alert{below the first reco bin entirely}.
    \item Fixed by switching to \texttt{trk\_range\_muon\_mom\_v}
      (\texttt{CC1mu1piXp.cxx}), which returns a momentum and exploits
      $m_\mu\!\approx\!m_\pi$.
  \end{itemize}
  \vspace{1mm}
  {\scriptsize Note: the proton-hypothesis range KE remains the \emph{correct} estimator
  for the \alert{proton} in the W/TKI subsample --- there the quantity wanted really is a
  kinetic energy, converted to momentum via $|\vec p_p|=\sqrt{T_p^2+2T_pm_p}$.\\[1mm]
  Resolution panels produced before the switch quote a $\sim\!21\%$ underestimate of
  $p_\pi$; they are to be regenerated with the current estimator.}
\end{frame}

\begin{frame}{Backup: W/TKI observable definitions}
  \small From the muon, charged-pion and leading-proton four-vectors above
  (framework STV tool):
  \begin{align*}
  W_{\pi p} &= \sqrt{(E_\pi+E_p)^2-|\vec p_\pi+\vec p_p|^2}\quad\text{(direct, no $\nu$ direction)}\\
  W_\mathrm{had} &= \sqrt{m_p^2+2m_p(E_\mathrm{cal}-E_\mu)-Q^2}\quad\text{(calorimetric)}\\
  \delta p_T &= |\vec p_T^{\,\mu}+\vec p_T^{\,\mathrm{had}}|,\quad
  \delta\alpha_T,\ \delta\phi_T\ \text{(TKI angles)},\quad p_n=\sqrt{\delta p_T^2+p_L^2}
  \end{align*}
  {\footnotesize with $\vec p^{\,\mathrm{had}}=\vec p_\pi+\vec p_p$,
  $E_\mathrm{cal}=E_\mu+E_\pi+T_p$, and the transverse components taken w.r.t.\ the beam
  ($\hat z$) axis. $E_b=24.78$~MeV argon removal energy enters $p_L$ (hence $p_n$).}
\end{frame}

  \vspace{1mm}
  {\scriptsize In $\cos\theta_\mu$ the $A_C$ smearing suppresses the peaked generators by
  $\times0.33$ but the smoother tune by much less, so NEUT and NuWro are pushed far from
  the data and $\chi^2$ explodes. In $\theta_\mu$, where the suppression is $\times0.76$,

  therefore mostly a variable-choice artefact, not model disagreement} --- a further
  argument for quoting $\theta$ rather than $\cos\theta$.}
\end{frame}
